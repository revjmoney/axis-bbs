# The native Linux program, axis: the board's own src/main.cpp, unchanged, on a Linux host layer
# (src/platform/linux) that provides the Arduino and ESP-IDF calls it makes. The screens are LovyanGFX's, from
# the board's PlatformIO copy: cmake -DLOVYANGFX_DIR=<.pio/libdeps/cyd/LovyanGFX>. Without it, axis is skipped.
if(UNIX AND NOT APPLE)
  set(LOVYANGFX_DIR "" CACHE PATH "LovyanGFX source tree (the board's PlatformIO copy, .pio/libdeps/cyd/LovyanGFX)")
  find_package(SDL2 QUIET)
  if(SDL2_FOUND)
    set(AXIS_WINDOW_DEFAULT ON)
  else()
    set(AXIS_WINDOW_DEFAULT OFF)
  endif()
  option(AXIS_WINDOW "axis opens a window showing the board's screens (needs the SDL2 dev files)" ${AXIS_WINDOW_DEFAULT})

  if(NOT LOVYANGFX_DIR OR NOT EXISTS "${LOVYANGFX_DIR}/src/LovyanGFX.hpp")
    message(STATUS "axis: LOVYANGFX_DIR is not a LovyanGFX tree, so the axis program is not built")
  else()
    # LovyanGFX for the PC. Its platform is picked by a macro: the SDL window with AXIS_WINDOW, otherwise its
    # Linux framebuffer platform, which needs nothing installed (axis never gives it a framebuffer panel: with no
    # window it runs as a headless board). Set explicitly, so installing SDL later cannot switch it silently.
    file(GLOB LGFX_SOURCES CONFIGURE_DEPENDS
        ${LOVYANGFX_DIR}/src/lgfx/Fonts/efont/*.c
        ${LOVYANGFX_DIR}/src/lgfx/Fonts/IPA/*.c
        ${LOVYANGFX_DIR}/src/lgfx/utility/*.c
        ${LOVYANGFX_DIR}/src/lgfx/v1/*.cpp
        ${LOVYANGFX_DIR}/src/lgfx/v1/panel/*.cpp
        ${LOVYANGFX_DIR}/src/lgfx/v1/touch/*.cpp)
    add_library(lgfx_host STATIC ${LGFX_SOURCES})
    target_include_directories(lgfx_host SYSTEM PUBLIC ${LOVYANGFX_DIR}/src)
    target_compile_definitions(lgfx_host PUBLIC LGFX_USE_V1=)  # empty, as board.h defines it
    target_compile_options(lgfx_host PRIVATE -w)  # a third-party library: its warnings are not ours to fix
    # one section per font and function, so the link drops what axis never uses (its CJK fonts are about 44 MB)
    target_compile_options(lgfx_host PRIVATE -ffunction-sections -fdata-sections)
    if(AXIS_WINDOW)
      find_package(SDL2 REQUIRED)
      target_compile_definitions(lgfx_host PUBLIC LGFX_SDL)
      target_link_libraries(lgfx_host PUBLIC SDL2::SDL2)
    else()
      target_compile_definitions(lgfx_host PUBLIC LGFX_LINUX_FB)
    endif()
    target_link_libraries(lgfx_host PUBLIC pthread)

    # The core again, built the way axis is: the cluster with the host's hooks (AXIS_CLUSTER=1 AXIS_NATIVE=1, see
    # src/cluster/cluster.h) and no SD pool (AXIS_SD_POOL stays unset, so 0), every other feature at the board's
    # default (bbs/features.h). axiscore above keeps the test flags. AXIS_PC adds the release-check row on this
    # program only; the firmware and the host tests never set it.
    add_library(axiscore_native STATIC ${AXISCORE_SOURCES})
    target_include_directories(axiscore_native PUBLIC src dosemu third_party/i8086 PRIVATE third_party/sha256 third_party/monocypher third_party/miniz)
    target_compile_definitions(axiscore_native PUBLIC AXIS_CLUSTER=1 AXIS_NATIVE=1 AXIS_BENCH=1 AXIS_PC=1)
    target_compile_options(axiscore_native PRIVATE -w)  # as the board builds it: the core's warnings are its own

    add_executable(axis
        src/main.cpp
        src/platform/linux/main_linux.cpp
        src/platform/linux/updatecheck.cpp
        src/platform/linux/updatecheck_linux.cpp
        src/platform/linux/arduino_core.cpp
        src/platform/linux/board.cpp
        src/platform/linux/wifi.cpp
        src/platform/linux/web.cpp
        src/platform/linux/storage.cpp
        src/platform/linux/paths.cpp
        src/platform/linux/fs.cpp
        src/platform/linux/fs_remap.cpp
        src/platform/linux/preferences.cpp
        src/platform/linux/serialfiles.cpp
        src/platform/linux/heapscope.cpp
        src/platform/linux/hostconf.cpp
        src/platform/linux/hostnet.cpp
        src/platform/linux/modemline.cpp
        src/platform/linux/startmode.cpp
        src/platform/linux/netgate.cpp
        src/platform/linux/clusterhost.cpp
        src/platform/linux/cardmirror.cpp
        src/platform/linux/mirrorhost.cpp
        src/platform/linux/handback.cpp
        src/platform/linux/nativecmd.cpp
        src/platform/cyd/clusternet.cpp    # the board's own: the cluster's UDP and link sockets through WiFiUDP and WiFiClient
        src/platform/cyd/config_store.cpp  # the board's own: LittleFS is the data folder's flash/ (fs.cpp)
        src/platform/cyd/sdfiles.cpp       # the board's own: its /sd/files paths go through fs_remap.cpp
        src/platform/cyd/screen.cpp        # the board's own: pure LovyanGFX, nothing to port
        src/platform/cyd/ansi_screen.cpp)
    if(AXIS_WINDOW)  # the window start mode: LovyanGFX's SDL panel as the board's screen
      target_sources(axis PRIVATE src/platform/linux/window.cpp)
    endif()
    # The board's screen files get Arduino.h (Serial, millis) through LovyanGFX's Arduino build on the ESP32.
    set_source_files_properties(src/platform/cyd/screen.cpp src/platform/cyd/ansi_screen.cpp
        PROPERTIES COMPILE_OPTIONS "-include;Arduino.h")
    # main.cpp is the board's file, built with -w as the board builds it (src/platform/linux keeps its warnings).
    set_source_files_properties(src/main.cpp PROPERTIES COMPILE_OPTIONS "-w")
    set_source_files_properties(src/platform/cyd/clusternet.cpp PROPERTIES COMPILE_OPTIONS "-w")  # the board's too
    target_include_directories(axis BEFORE PRIVATE src/platform/linux/arduino)
    target_include_directories(axis PRIVATE src/platform/cyd third_party/monocypher)
    target_compile_definitions(axis PRIVATE AXIS_CLUSTER=1 AXIS_NATIVE=1 AXIS_PC=1 LGFX_USE_V1= AXIS_WINDOW=$<BOOL:${AXIS_WINDOW}>)
    target_link_libraries(axis PRIVATE axiscore_native lgfx_host)
    target_link_options(axis PRIVATE -Wl,--gc-sections)
    # The board's paths (/sd, /littlefs) become the data folder's at link time: every call below, from main.cpp,
    # the core or the platform files, goes through src/platform/linux/fs_remap.cpp. Only this executable is wrapped.
    foreach(fn fopen fopen64 freopen open open64 __open_2 __open64_2 stat stat64 lstat lstat64 mkdir rename remove
            opendir unlink rmdir access truncate statvfs utime)
      target_link_options(axis PRIVATE "-Wl,--wrap=${fn}")
    endforeach()
    # configLoad and configSave (the board's config_store.cpp) map the board's telnet port 23 to lan_port and back,
    # so the port main.cpp shows is the one axis listens on (src/platform/linux/hostnet.cpp, at its end).
    target_link_options(axis PRIVATE "-Wl,--wrap=_Z10configLoadv" "-Wl,--wrap=_Z10configSaveRKN4axis3bbs9BbsConfigE")
  endif()
endif()

# Packaging for axispc (Project 1a, Task 6): install rules and a CPack Debian package. Only axispc and
# its installer files are installed, so cpack never pulls in axis_tests, the tools or the axis program.
# This block adds no compile step, so the board build (which never includes this file, it uses the
# ESP-IDF branch of CMakeLists.txt) and the host tests are unaffected. axispc uses no LovyanGFX, so the
# package builds whether or not LOVYANGFX_DIR is set.
if(UNIX AND NOT APPLE AND TARGET axispc)
  # The package version is kAxisVersion in src/bbs/version.h, the one place the version lives.
  file(READ ${CMAKE_CURRENT_LIST_DIR}/src/bbs/version.h _axispc_version_h)
  string(REGEX MATCH "kAxisVersion[ \t]*=[ \t]*\"([0-9][0-9.]*)\"" _axispc_vmatch "${_axispc_version_h}")
  if(NOT CMAKE_MATCH_1)
    message(WARNING "axispc packaging: no version read from src/bbs/version.h (kAxisVersion); the .deb is not configured")
  endif()
endif()
# Skip packaging (not an error) if the version could not be read, so a plain host-test configure
# never fails over it; the .deb is only configured when there is a version.
if(UNIX AND NOT APPLE AND TARGET axispc AND CMAKE_MATCH_1)
  set(AXISPC_VERSION "${CMAKE_MATCH_1}")

  include(GNUInstallDirs)
  # Package directories are the usual drwxr-xr-x (set before the install() rules so it takes effect).
  set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS
      OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  install(TARGETS axispc RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})  # /usr/bin/axispc
  install(FILES installer/pc/axispc.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES installer/pc/axispc.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
  install(FILES installer/pc/axispc.service DESTINATION lib/systemd/system)  # /usr/lib/systemd/system
  install(FILES installer/pc/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/axispc)

  set(CPACK_GENERATOR "DEB")
  set(CPACK_PACKAGE_NAME "axispc")
  set(CPACK_PACKAGE_VERSION "${AXISPC_VERSION}")
  set(CPACK_PACKAGE_CONTACT "Rev. J. Money <axis@jmscnc.com>")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "AXIS BBS, the native PC bulletin board program")
  set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Rev. J. Money <axis@jmscnc.com>")
  set(CPACK_DEBIAN_PACKAGE_SECTION "comm")
  # Depends only on libc6 and libstdc++6: axispc links the BBS core statically and uses pthread (in libc6).
  # Hardcoded, so the automatic shared library scan is off (it would add versioned entries).
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS OFF)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6, libstdc++6")
  # Names the file axispc_<version>_<debarch>.deb. The architecture is dpkg's on a native build, or
  # cpack -D CPACK_DEBIAN_PACKAGE_ARCHITECTURE=armhf for the 32-bit ARM cross build.
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
  include(CPack)
endif()
